from celery import Celery
from database import supabase
import time

celery_app = Celery(
  'document_processor', #Name of the Celery app
  broker='redis://localhost:6379/0',  #Where tasks are queued
  backend='redis://localhost:6379/0'  #Where results are stored (optional, can be used for tracking task status)
)

@celery_app.task
def process_document(document_id: str):
    """
      Simple test task
    """

    # step 1: update status to processing
    supabase.table("project_documents").update({
        "processing_status": "processing"
    }).eq("id", document_id).execute()

    print(f"Processing document: {document_id}")

    # step 2: Actual work (e.g., chunking, extract text, generate embeddings, etc.)
    time.sleep(5)  # Simulate time-consuming processing

    # step 3: update status to completed or failed based on the outcome
    supabase.table("project_documents").update({
        "processing_status": "completed"
    }).eq("id", document_id).execute()  

    print(f"Completed processing document: {document_id}")

    return {
        "status": "success",
        "document_id": document_id
    }
